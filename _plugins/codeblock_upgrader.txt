def return_raw(content)
#  return unless item.output_ext == '.html'
  
 # puts "=== DEBUG: Processing #{item.relative_path} ==="
  content = content.dup
  
  # SHOW RAW STRUCTURE
  puts "=== RAW SNIPPET (first 500 chars): ==="
  puts content[0..500]
  puts "=== ALL <pre> tags: ==="
  puts content.scan(/<pre[^>]*>/).join("\n")
  puts "=== ALL div classes: ==="
  puts content.scan(/class="([^"]*)"/).flatten.join("\n")
  
  doc = Nokogiri::HTML(content)

  puts "- code: #{doc.css('code').count}"

module Jekyll
  class ChirpyFinalizer < Generator
    safe true
    priority :low

    def generate(site)
      # Process after Jekyll finishes all rendering
      site.pages.concat(site.posts.docs).each do |item|
        next unless item.output_ext == '.html'
        
        # Read the already-rendered HTML file from _site
        site_path = File.join(site.dest, item.url.sub(/index$/, ''))
        next unless File.exist?(site_path)
        
        puts "=== DEBUG: Processing #{item.relative_path} (#{site_path}) ==="
        
        html = File.read(site_path)
        doc = Nokogiri::HTML(html)
        containers = doc.css('div.highlighter-rouge')
        
        puts "=== DEBUG: Found #{containers.count} highlighter-rouge blocks ==="
        next unless containers.any?
        
        containers.each_with_index do |container, i|
          lang = container['class'].match(/language-(\w+)/)[1]&.upcase || 'CODE'
          highlight = container.css('div.highlight').first
          
          puts "=== Block #{i+1}: #{lang} ==="
          
          chirpy_wrapper = Nokogiri::XML::DocumentFragment.parse(<<~HTML)
            <div class="code-block" data-lang="#{lang}">
              <div class="code-header">
                <div class="code-header-left">
                  <i class="fas fa-code fa-fw small"></i>
                  <span class="code-lang">#{lang}</span>
                </div>
                <button class="copy-btn" type="button" aria-label="Copy code">
                  <i class="far fa-clipboard"></i>
                  <span class="copy-label">Copy</span>
                </button>
              </div>
              #{highlight.to_html}
            </div>
          HTML
          
          container.replace(chirpy_wrapper)
        end
        
        # Overwrite the final file
        File.write(site_path, doc.to_html)
        puts "=== Saved #{doc.css('.code-block').count} Chirpy blocks ==="
      end
    end
  end
end
