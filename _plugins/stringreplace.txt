require 'nokogiri'


module Jekyll
  class StringReplacer < Converter
    safe true
    priority :high

    def matches(ext)
      ext =~ /^\.html$/
    end

    def convert(input)
      puts "=== TEST: StringReplacer running ==="
      result = input.gsub('Hello', '<span style="color:red;background:yellow">HELLO WORLD</span>')
      puts "=== TEST: Replaced #{input.scan('Hello').count} → #{result.scan('HELLO WORLD').count} ==="
      result
    end

    def output_ext(ext)
      ext
    end
  end
end
